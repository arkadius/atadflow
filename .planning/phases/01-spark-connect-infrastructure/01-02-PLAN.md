```java
package io.atadflow.health;

import io.quarkus.logging.Log;
import io.quarkus.runtime.Startup;
import jakarta.enterprise.context.ApplicationScoped;

import java.io.BufferedReader;
import java.io.InputStreamReader;

@ApplicationScoped
@Startup
public class PythonHealthCheck {

    public PythonHealthCheck() {
        checkPython();
        checkPySpark();
        Log.info("Python health check passed: Python and PySpark are available");
    }

    private void checkPython() {
        try {
            ProcessBuilder pb = new ProcessBuilder("python", "--version");
            pb.redirectErrorStream(true);
            Process process = pb.start();

            try (BufferedReader reader = new BufferedReader(new InputStreamReader(process.getInputStream()))) {
                String version = reader.readLine();
                int exitCode = process.waitFor();

                if (exitCode != 0 || version == null) {
                    throw new RuntimeException("Python not found or not working. Exit code: " + exitCode);
                }

                Log.info("Python version: " + version);
            }
        } catch (Exception e) {
            throw new RuntimeException("Python health check failed. Ensure Python is installed and on PATH.", e);
        }
    }

    private void checkPySpark() {
        try {
            ProcessBuilder pb = new ProcessBuilder("pyspark", "--version");
            pb.redirectErrorStream(true);
            Process process = pb.start();

            try (BufferedReader reader = new BufferedReader(new InputStreamReader(process.getInputStream()))) {
                StringBuilder output = new StringBuilder();
                String line;
                while ((line = reader.readLine()) != null) {
                    output.append(line).append("\n");
                }

                int exitCode = process.waitFor();

                if (exitCode != 0) {
                    throw new RuntimeException("PySpark not found or not working. Exit code: " + exitCode);
                }

                // Extract version from output (typically contains "version X.Y.Z")
                String versionLine = output.toString().lines()
                    .filter(l -> l.contains("version"))
                    .findFirst()
                    .orElse("PySpark installed");

                Log.info("PySpark check: " + versionLine.trim());
            }
        } catch (Exception e) {
            throw new RuntimeException("PySpark health check failed. Ensure PySpark is installed and on PATH.", e);
        }
    }
}
```

**Why this implementation:**
- `@Startup` ensures the check runs at application startup (before serving requests)
- `@ApplicationScoped` makes it a CDI bean
- Throws RuntimeException if Python or PySpark is missing → Quarkus startup fails fast with clear error
- Logs version info on success → visible in startup logs
- Uses ProcessBuilder to execute shell commands
